\begin{frame}{Side Effects}
    \showmytikz{../shared/tikz/tikz-PM-side-effects}
    \mycomment{
    
    \begin{itemize}
        \item If you improve $A$, you improve $R$
        \item But how does that influence $B$?
        \item $A \uparrow\; \rightarrow\; R \uparrow$
        \item $\left( A \; \uparrow \; \rightarrow R\uparrow \right)\; \land \; \overline{R} \rightarrow B \downarrow$
        \item `change one thing in a system, other things may change too!'
    \end{itemize}
    }
\end{frame}

\begin{frame}{Partial vs. Total Differential}
    
    \mytext{
\parbox{0.6\textwidth}{
        \begin{equation}
        \frac{\partial R}{\partial A} \; \neq \; \frac{dR}{dA} 
    \end{equation}
    \begin{equation}
        \left[ \frac{\partial R}{\partial A} > 0 \;  \land \; \frac{\partial R}{\partial B} > 0 \right] \; \nrightarrow \; \frac{dR}{dA} > 0
    \end{equation}
}
    }
\end{frame}

\begin{frame}{Different Understanding of "Change Management"}
    
    \mytext{
    
    
    \begin{itemize}
        \item Project Management:
        \item[] \textit{how to manage changes to the plan}
        \begin{itemize}
            \item   log change, documentation
            \item change the plan (software?)
            \item communicate / enforce changes
            \item cascading consequences
        \end{itemize}
        \item Organisational Change:
        \item[] \textit{manage the development / change of an organisation}
        \begin{itemize}
            \item change of organisational structure 
            \item adaption of organisation to change
            \item often: "de-procrustinate"
            \item change "business as usual"
            \item (huge!) challenges for people
        \end{itemize}
    \end{itemize}
    }
\end{frame}
    
\begin{frame}{Changing is Hard}
    
    \mytext{
    \begin{itemize}
        \item Change a plan is hard
        \item[] -- but things/objects do not discuss
        \item Change people is harder
        \item[] -- convincing takes time
        \item[] -- communication is key
        \item[] -- "walk away" is easier, but is it better?
        \item Every plan involves people,
        \item[] so the two understandings of change management relate:
        \item[] -- plan change affects organisation
        \item[] -- organisational change changes plans
    \end{itemize}
    }
\end{frame}

\begin{frame}{Change Other People}
    
    \mytext{
    \begin{itemize}
        \item have those above you to change is hard
        \item[] -- "management from below"
        \item[] -- introduce ideas strategically
        \item[] -- "elevator pitch"
        \item have those below you to change is hard
        \item[] -- "leadership"
        \item[] -- imperative or suggestive
        \item[] -- consult and decide
        \item have those on the same level as you to change is hard
        \item[] -- "influencing"
        \item[] -- rarely a change will not be changed itself
    \end{itemize}
    }
