    you always take the best decision 
    
    \emph{given your ability}

    \medskip

    }

\begin{frame}{Search theory, Secretary Problem}
    
    \mytext{
    Choose between some things about which you know almost nothing.

    \begin{description}
        \item[phase 1] gather information
        \item[phase 2] take the next one that is better than the ones before
    \end{description}
    
    Optimal time to switch from "search" to "jump":
    \begin{itemize}
        \item reject the first $n/e$ applicants ($e=2.718\dots$); 
        \item jump at next better one;
        \item chance to pick best is jump best is ca.\ 37\%
        \item Or: use Gittins intex
    \end{itemize}
    
    \cite{algoliveby}
    }
\end{frame}

        \item We see risks more than rewards
        \item Fear of failure
        \item Fear of the unknown
        \item Paralysis by analysis

    }
