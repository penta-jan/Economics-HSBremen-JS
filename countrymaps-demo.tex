\section*{States and provinces}
Available for Germany, India, Canada, USA, Brazil, China, South Africa, Nigeria, Mexico, Australia, Argentina and Russia.

\begin{minipage}[t]{.32\linewidth}
\begin{lstlisting}
\drawmap{germany,bremen}
\end{lstlisting}
\drawmap[width=4.5cm]{germany,bremen[fill=red!50]}
\end{minipage}\hfill
\begin{minipage}[t]{.32\linewidth}
\begin{lstlisting}
\drawmap[states]
  {germany, bayern[fill=blue!20]}
\end{lstlisting}
\drawmap[width=4.5cm, states]{germany, bayern[fill=blue!20]}
\end{minipage}\hfill
\begin{minipage}[t]{.32\linewidth}
\begin{lstlisting}
\drawmap[context, margin=.3,
  labels]{bremen}
\end{lstlisting}
\drawmap[width=4.5cm, context, margin=.3, labels]{bremen}
\end{minipage}
