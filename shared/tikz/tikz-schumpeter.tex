\pgfmathdeclarefunction{oldtech}{1}{\pgfmathparse{30*(1-sin(deg(#1-pi/2)))}}
\pgfmathdeclarefunction{newtech}{1}{\pgfmathparse{65*(1+sin(deg(#1-pi/2)))}}


% Analytic intersection: oldtech(t)=newtech(t)
% => 50 - 50 sin(t-pi/2) = 65 + 65 sin(t-pi/2)
% => sin(t-pi/2) = -3/23
\pgfmathsetmacro{\sinval}{-3/23}

\pgfmathsetmacro{\angledeg}{asin(\sinval)}      % pgfmath asin returns degrees
\pgfmathsetmacro{\thetarad}{\angledeg*pi/180}   % convert to radians
\pgfmathsetmacro{\tstar}{pi/2+\thetarad}
\pgfmathsetmacro{\ystar}{50*(1-\sinval)}

    \path [name intersections={of=oldtech and newtech,by=E}];
